$duration = ffprobe -i song1.mp3 -show_entries format=duration -v quiet -of csv="p=0"
$start = Get-Random -Minimum 0 -Maximum ([math]::Floor($duration - 10))
ffmpeg -ss $start -t 10 -i song1.mp3 -c:a pcm_s16le clip1.wav
